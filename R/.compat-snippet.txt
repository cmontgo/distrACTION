
# ---- ggplot2 compatibility helper for geom_segment linewidth/size ----

    do.call(geom_segment, c(list(mapping = aes(x=HigherSegment, y=0, xend=HigherSegment, yend=HigherSegmentLength, linetype=QuantileLabel),colour = Color[2], alpha = QuantileAlphaHigh), segment_line_args))+
    scale_linetype_manual(values=TypeOfLine)
}
